cmake_minimum_required(VERSION 3.21)
project(YggGuard VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network)
add_executable(yggguard src/main.cpp)
target_link_libraries(yggguard PRIVATE Qt6::Widgets Qt6::Network)
install(TARGETS yggguard RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    add_executable(yggguard-ui-smoke tests/ui_smoke.cpp)
    target_link_libraries(yggguard-ui-smoke PRIVATE Qt6::Widgets Qt6::Network)
    add_test(NAME ui-smoke COMMAND yggguard-ui-smoke)
    set_tests_properties(ui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
endif()
